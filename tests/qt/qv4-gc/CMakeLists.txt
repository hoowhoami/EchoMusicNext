cmake_minimum_required(VERSION 3.28)
project(EchoQv4GcRegression LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)

find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Core Gui Qml QmlPrivate Quick Test)
